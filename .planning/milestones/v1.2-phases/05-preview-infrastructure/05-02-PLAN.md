Implement using JavaScriptCore for synchronous markdown-it execution:

```swift
// MarkdownEditor/MarkdownRenderer.swift
import Foundation
import JavaScriptCore

final class MarkdownRenderer {

    // MARK: - Singleton JS context (created once, reused)

    private static let jsContext: JSContext? = {
        guard let context = JSContext() else { return nil }

        // Load markdown-it.min.js from bundle
        if let url = Bundle.main.url(forResource: "markdown-it.min", withExtension: "js"),
           let script = try? String(contentsOf: url, encoding: .utf8) {
            context.evaluateScript(script)
        }

        // Set up markdown-it with tables enabled (GFM table syntax)
        context.evaluateScript("""
            var md = window.markdownit ? window.markdownit({ html: false, linkify: false, typographer: false }) : null;
            if (!md && typeof markdownit !== 'undefined') {
                md = markdownit({ html: false, linkify: false, typographer: false });
            }
            function renderToHTML(markdownText) {
                if (!md) return '<p>Error: markdown-it not loaded</p>';
                return md.render(markdownText);
            }
        """)

        return context
    }()

    // MARK: - CSS content (read from bundle once)

    private static let cssContent: String = {
        guard let url = Bundle.main.url(forResource: "markdown-preview", withExtension: "css"),
              let css = try? String(contentsOf: url, encoding: .utf8) else {
            return "/* markdown-preview.css not found */"
        }
        return css
    }()

    // MARK: - JS helper content (read from bundle once)

    private static let jsHelperContent: String = {
        guard let url = Bundle.main.url(forResource: "markdown-preview", withExtension: "js"),
              let js = try? String(contentsOf: url, encoding: .utf8) else {
            return "/* markdown-preview.js not found */"
        }
        return js
    }()

    // MARK: - Public API

    /// Generates a complete HTML document string from a markdown input.
    /// The returned string is ready to pass to WKWebView.loadHTMLString(_:baseURL:).
    /// Uses JavaScriptCore to run markdown-it synchronously for immediate HTML output.
    static func generateHTML(from markdown: String) -> String {
        // Render markdown → HTML fragment using JavaScriptCore
        let bodyHTML: String
        if let context = jsContext,
           let result = context.evaluateScript("renderToHTML(\(markdown.jsonEncoded()))"),
           !result.isUndefined, !result.isNull {
            bodyHTML = result.toString() ?? ""
        } else {
            bodyHTML = "<p>\(markdown.escapingHTMLEntities())</p>"
        }

        return """
        <!DOCTYPE html>
        <html lang="en">
        <head>
            <meta charset="UTF-8">
            <meta name="viewport" content="width=device-width, initial-scale=1.0">
            <style>
            \(cssContent)
            </style>
        </head>
        <body>
            <div id="content">\(bodyHTML)</div>
        </body>
        </html>
        """
    }
}

// MARK: - String helpers

private extension String {
    /// Encodes a Swift String as a JavaScript string literal (JSON-encoded).
    func jsonEncoded() -> String {
        let data = try? JSONSerialization.data(withJSONObject: self)
        return data.flatMap { String(data: $0, encoding: .utf8) } ?? "\"\""
    }

    /// Escapes HTML entities for safe inline insertion.
    func escapingHTMLEntities() -> String {
        self
            .replacingOccurrences(of: "&", with: "&amp;")
            .replacingOccurrences(of: "<", with: "&lt;")
            .replacingOccurrences(of: ">", with: "&gt;")
            .replacingOccurrences(of: "\"", with: "&quot;")
    }
}
```

NOTES FOR EXECUTOR:
- `markdown-it.min.js` must be present in Copy Bundle Resources before this runs. If the file is missing, the JSContext silently produces no markdown-it; tests will fail with "Error: markdown-it not loaded".
- JSContext uses the global scope differently from a browser — `window` object does not exist. Use `markdownit(...)` directly (not `window.markdownit(...)`). The evaluateScript sets up `md` using both `window.markdownit` (for browser) and `markdownit` (for JSContext). Ensure the setup script handles both.
- The markdown-it library v14.1.1 must be downloaded from https://cdn.jsdelivr.net/npm/markdown-it@14.1.1/dist/markdown-it.min.js and added to the Xcode project before running tests.
- highlight.min.js is NOT loaded in JSContext (syntax highlighting requires DOM; deferred to WKWebView in Plan 03). Tests do not assert on highlight.js behavior.
- Tables require the `markdown-it` tables plugin OR built-in GFM table support. markdown-it v14.1.1 does NOT have built-in table support — it needs the `markdown-it-table` plugin. Include this in the JSContext setup by also bundling `markdown-it-table.min.js` OR use the alternative `markdownit` initialization option `{linkify: false}` with the table plugin. If the table plugin is not bundled, the table test will fail — add `markdown-it-table` download step: https://cdn.jsdelivr.net/npm/@gerhobbelt/markdown-it-table@latest/dist/markdown-it-table.min.js or use the simpler `markdown-it` with GFM mode.
