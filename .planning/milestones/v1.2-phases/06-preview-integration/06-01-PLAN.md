   ```swift
   @ViewBuilder
   private func editorView(for doc: MarkdownDocument) -> some View {
       if isPreviewMode {
           PreviewView(markdownString: doc.text)
               .ignoresSafeArea()
       } else {
           MarkdownTextEditor(
               text: Binding(
                   get: { doc.text },
                   set: { newValue in
                       doc.text = newValue
                       scheduleSave(for: doc)
                   }
               ),
               isHighlightingEnabled: !highlightingDisabled
           )
           .ignoresSafeArea(.keyboard)
       }
   }
   ```

